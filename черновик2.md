db.mk_products.aggregate([
    // Шаг 1: Разворачиваем массив PROIZ, чтобы работать с каждым произведением
    { 
        $unwind: "$PROIZ" 
    },
    // Шаг 2: Разворачиваем массив TIRAJ, чтобы работать с каждым тиражом
    { 
        $unwind: { path: "$TIRAJ", preserveNullAndEmptyArrays: true } 
    },
    // Шаг 3: Группировка по GUID произведения
    {
        $group: {
            _id: "$PROIZ.GUID", // GUID произведения как ключ группировки
            title: { $first: "$PROIZ.NAME" }, // Название произведения
            cover: { $first: "$NOMCODE" }, // Обложка
            created_at: { $first: "$created_at" },
            updated_at: { $first: "$updated_at" },
            characteristics: {
                $first: {
                    isbn: "$ISBN",
                    series: { $ifNull: ["$SERIE.NAME", ""] },
                    format: { $ifNull: ["$PEREPLET_TYPE.NAME", ""] },
                    release_date: null,
                    pages: 0,
                    material: null,
                    weight: null,
                    thickness: null,
                    time_to_read: null
                }
            },
            tiraj: {
                $push: {
                    tiraj_code: "$TIRAJ.TIRAJ_CODE",
                    tiraj_type: "$TIRAJ.TIRAJ_TYPE",
                    ekzempl_v_tiraje: "$TIRAJ.EKZEMPL_V_TIRAJE",
                    fakticheskaya_data: "$TIRAJ.FAKTICHESKAYA_DATA",
                    plan_date: "$TIRAJ.PLAN_DATE"
                }
            }
        }
    },
    // Шаг 4: Финальный проект с форматированием
    {
        $project: {
            _id: 0,
            id: "$_id",
            created_at: 1,
            updated_at: 1,
            cover: 1,
            title: 1,
            characteristics: 1,
            tiraj: 1
        }
    },
    // Шаг 5: Оборачиваем в итоговый массив
    {
        $group: {
            _id: null,
            data: { $push: "$$ROOT" }
        }
    },
    {
        $project: {
            success: { $literal: true },
            data: ["$data"]
        }
    }
]);

      as: "WorksList"

  },
  {
    $unwind: "$WorksList"
  },
  {
    $lookup: {
      from: "mk_products",
      localField: "WorksList.products.GUID",
      foreignField: "GUID",
      as: "TirageList"
    }
  },
  {
    $unwind: "$TirageList"
  },
  {
    $unwind: "$TirageList.TIRAJ"
  },
  {
    $project: {
      id: "$TirageList._id",
      created_at: "$TirageList.created_at",
      updated_at: "$TirageList.updated_at",
      cover: { $ifNull: ["$TirageList.COVERS.cover1.NOMCODE", ""] },

        format: "$TirageList.PEREPLET_TYPE.NAME",
