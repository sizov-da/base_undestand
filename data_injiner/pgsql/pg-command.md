### Подключение к базе данных

Чтобы подключиться к конкретной базе данных, используйте следующую команду:
```bash
psql -h localhost -p 5432 -U postgres -d имя_базы_данных
```

Например, для подключения к базе данных `crypto_wallet`:
```bash
psql -h localhost -p 5432 -U postgres -d crypto_wallet
```

чтобы вывести данные из таблицы `users`, выполните:
```sql
SELECT * FROM users;
```
выввести только 10 записей:
```sql
SELECT * FROM users LIMIT 10;
```
сохранить дамп базы данных в файл:
```bash
pg_dump -h localhost -p 5432 -U postgres -d имя_базы_данных > имя_файла.sql
```
### Основные команды PostgreSQL
```sql
-- Создать новую базу данных
CREATE DATABASE имя_базы_данных;
-- Удалить базу данных
DROP DATABASE имя_базы_данных;
-- Создать новую таблицу
CREATE TABLE имя_таблицы (
    столбец1 тип_данных,
    столбец2 тип_данных,
    ...
);
-- Удалить таблицу
DROP TABLE имя_таблицы;
-- Добавить новый столбец в таблицу
ALTER TABLE имя_таблицы ADD COLUMN новый_столбец тип_данных;
-- Удалить столбец из таблицы
ALTER TABLE имя_таблицы DROP COLUMN столбец;
-- Изменить тип данных столбца
ALTER TABLE имя_таблицы ALTER COLUMN столбец TYPE новый_тип;
-- Вставить данные в таблицу
INSERT INTO имя_таблицы (столбец1, столбец2, ...)
VALUES (значение1, значение2, ...);
-- Обновить данные в таблице
UPDATE имя_таблицы
SET столбец1 = новое_значение1, столбец2 = новое_значение2
WHERE условие;
-- Удалить данные из таблицы
DELETE FROM имя_таблицы WHERE условие;
-- Выбрать данные из таблицы
    
SELECT столбец1, столбец2, ...
